table node
  column id uuid not null
  column type text not null
  unique node_id_type_unique (id, type)
table node_version
  column id uuid not null
  column version bigint not null
  column action text not null
  column timestamp timestamp with time zone not null
  column actor text
  primary key node_version_pkey (id, version)
  foreign key node_version_id_node_id_fk (id) → node (id) on update no action on delete cascade
  check node_version_action_valid: "ontology"."node_version"."action" IN ('create', 'update', 'delete')
table works
  column id uuid not null
  column type text not null
  foreign key works_id_node_id_fk (id) → node (id) on update no action on delete cascade
  foreign key works_id_type_fk (id, type) → node (id, type) on update no action on delete no action
  unique works_id_type_unique (id, type)
  check works_type_valid: "ontology"."works"."type" IN ($1, $2, $3) -- novel, story, poem
table works_version
  column id uuid not null
  column version bigint not null
  column title text not null
  primary key works_version_pkey (id, version)
  foreign key works_version_id_works_id_fk (id) → works (id) on update cascade on delete cascade
  foreign key works_version_node_version_fk (id, version) → node_version (id, version) on update no action on delete cascade

view novel_view
  select "ontology"."node_version"."id", "ontology"."node_version"."version", "ontology"."node_version"."action", "ontology"."works_version"."title", "ontology"."novel_version"."word_count", "ontology"."node_version"."timestamp", "version_zero"."timestamp" as "created_at" from "ontology"."novel_version" inner join "ontology"."node_version" on (("ontology"."novel_version"."id" = "ontology"."node_version"."id") and ("ontology"."novel_version"."version" = "ontology"."node_version"."version")) inner join "ontology"."node" on "ontology"."novel_version"."id" = "ontology"."node"."id" inner join "ontology"."works_version" on (("ontology"."novel_version"."id" = "ontology"."works_version"."id") and ("ontology"."novel_version"."version" = "ontology"."works_version"."version")) inner join "ontology"."node_version" "version_zero" on (("ontology"."novel_version"."id" = "version_zero"."id") and ("version_zero"."version" = 0)) where not exists (select 1 from "ontology"."node_version" "newer" where (("newer"."id" = "ontology"."novel_version"."id") and ("newer"."version" > "ontology"."novel_version"."version")))

view works_view
