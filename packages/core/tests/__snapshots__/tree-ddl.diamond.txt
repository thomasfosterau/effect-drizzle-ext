table node
  column id uuid not null
  column type text not null
  unique node_id_type_unique (id, type)
table node_version
  column id uuid not null
  column version bigint not null
  column action text not null
  column timestamp timestamp with time zone not null
  column actor text
  primary key node_version_pkey (id, version)
  foreign key node_version_id_node_id_fk (id) → node (id) on update no action on delete cascade
  check node_version_action_valid: "ontology"."node_version"."action" IN ('create', 'update', 'delete')
table artefact
  column id uuid not null
  column type text not null
  foreign key artefact_id_node_id_fk (id) → node (id) on update no action on delete cascade
  foreign key artefact_id_type_fk (id, type) → node (id, type) on update cascade on delete no action
  unique artefact_id_type_unique (id, type)
  check artefact_type_valid: "ontology"."artefact"."type" IN ($1, $2, $3) -- printed, digital, ebook
table artefact_version
  column id uuid not null
  column version bigint not null
  column label text not null
  primary key artefact_version_pkey (id, version)
  foreign key artefact_version_id_artefact_id_fk (id) → artefact (id) on update cascade on delete cascade
  foreign key artefact_version_node_version_fk (id, version) → node_version (id, version) on update no action on delete cascade
table printed
  column id uuid not null
  column type text not null
  foreign key printed_id_artefact_id_fk (id) → artefact (id) on update no action on delete cascade
  foreign key printed_id_type_fk (id, type) → artefact (id, type) on update cascade on delete no action
  unique printed_id_type_unique (id, type)
  check printed_type_valid: "ontology"."printed"."type" IN ($1, $2) -- printed, ebook
table printed_version
  column id uuid not null
  column version bigint not null
  column press_run integer
  primary key printed_version_pkey (id, version)
  foreign key printed_version_id_printed_id_fk (id) → printed (id) on update cascade on delete cascade
  foreign key printed_version_artefact_version_fk (id, version) → artefact_version (id, version) on update no action on delete cascade
table digital
  column id uuid not null
  column type text not null
  foreign key digital_id_artefact_id_fk (id) → artefact (id) on update no action on delete cascade
  foreign key digital_id_type_fk (id, type) → artefact (id, type) on update cascade on delete no action
  unique digital_id_type_unique (id, type)
  check digital_type_valid: "ontology"."digital"."type" IN ($1, $2) -- digital, ebook
table digital_version
  column id uuid not null
  column version bigint not null
  column file_size integer
  primary key digital_version_pkey (id, version)
  foreign key digital_version_id_digital_id_fk (id) → digital (id) on update cascade on delete cascade
  foreign key digital_version_artefact_version_fk (id, version) → artefact_version (id, version) on update no action on delete cascade
table ebook
  column id uuid not null
  column type text not null
  foreign key ebook_id_printed_id_fk (id) → printed (id) on update no action on delete cascade
  foreign key ebook_id_type_fk (id, type) → printed (id, type) on update cascade on delete no action
  foreign key ebook_id_digital_id_fkey (id) → digital (id) on update no action on delete cascade
  foreign key ebook_id_type_digital_fk (id, type) → digital (id, type) on update cascade on delete no action
  unique ebook_id_type_unique (id, type)
  check ebook_type_valid: "ontology"."ebook"."type" IN ($1) -- ebook
table ebook_version
  column id uuid not null
  column version bigint not null
  column drm text
  primary key ebook_version_pkey (id, version)
  foreign key ebook_version_id_ebook_id_fk (id) → ebook (id) on update cascade on delete cascade
  foreign key ebook_version_printed_version_fk (id, version) → printed_version (id, version) on update no action on delete cascade
  foreign key ebook_version_digital_version_fk (id, version) → digital_version (id, version) on update no action on delete cascade
view printed_view
  select "ontology"."node_version"."id", "ontology"."node_version"."version", "ontology"."node_version"."action", "ontology"."artefact_version"."label", "ontology"."printed_version"."press_run", "ontology"."node_version"."timestamp", "version_zero"."timestamp" as "created_at" from "ontology"."printed_version" inner join "ontology"."node_version" on (("ontology"."printed_version"."id" = "ontology"."node_version"."id") and ("ontology"."printed_version"."version" = "ontology"."node_version"."version")) inner join "ontology"."node" on "ontology"."printed_version"."id" = "ontology"."node"."id" inner join "ontology"."artefact_version" on (("ontology"."printed_version"."id" = "ontology"."artefact_version"."id") and ("ontology"."printed_version"."version" = "ontology"."artefact_version"."version")) inner join "ontology"."node_version" "version_zero" on (("ontology"."printed_version"."id" = "version_zero"."id") and ("version_zero"."version" = 0)) where not exists (select 1 from "ontology"."node_version" "newer" where (("newer"."id" = "ontology"."printed_version"."id") and ("newer"."version" > "ontology"."printed_version"."version")))
view digital_view
  select "ontology"."node_version"."id", "ontology"."node_version"."version", "ontology"."node_version"."action", "ontology"."artefact_version"."label", "ontology"."digital_version"."file_size", "ontology"."node_version"."timestamp", "version_zero"."timestamp" as "created_at" from "ontology"."digital_version" inner join "ontology"."node_version" on (("ontology"."digital_version"."id" = "ontology"."node_version"."id") and ("ontology"."digital_version"."version" = "ontology"."node_version"."version")) inner join "ontology"."node" on "ontology"."digital_version"."id" = "ontology"."node"."id" inner join "ontology"."artefact_version" on (("ontology"."digital_version"."id" = "ontology"."artefact_version"."id") and ("ontology"."digital_version"."version" = "ontology"."artefact_version"."version")) inner join "ontology"."node_version" "version_zero" on (("ontology"."digital_version"."id" = "version_zero"."id") and ("version_zero"."version" = 0)) where not exists (select 1 from "ontology"."node_version" "newer" where (("newer"."id" = "ontology"."digital_version"."id") and ("newer"."version" > "ontology"."digital_version"."version")))
view ebook_view
  select "ontology"."node_version"."id", "ontology"."node_version"."version", "ontology"."node_version"."action", "ontology"."artefact_version"."label", "ontology"."printed_version"."press_run", "ontology"."digital_version"."file_size", "ontology"."ebook_version"."drm", "ontology"."node_version"."timestamp", "version_zero"."timestamp" as "created_at" from "ontology"."ebook_version" inner join "ontology"."node_version" on (("ontology"."ebook_version"."id" = "ontology"."node_version"."id") and ("ontology"."ebook_version"."version" = "ontology"."node_version"."version")) inner join "ontology"."node" on "ontology"."ebook_version"."id" = "ontology"."node"."id" inner join "ontology"."artefact_version" on (("ontology"."ebook_version"."id" = "ontology"."artefact_version"."id") and ("ontology"."ebook_version"."version" = "ontology"."artefact_version"."version")) inner join "ontology"."printed_version" on (("ontology"."ebook_version"."id" = "ontology"."printed_version"."id") and ("ontology"."ebook_version"."version" = "ontology"."printed_version"."version")) inner join "ontology"."digital_version" on (("ontology"."ebook_version"."id" = "ontology"."digital_version"."id") and ("ontology"."ebook_version"."version" = "ontology"."digital_version"."version")) inner join "ontology"."node_version" "version_zero" on (("ontology"."ebook_version"."id" = "version_zero"."id") and ("version_zero"."version" = 0)) where not exists (select 1 from "ontology"."node_version" "newer" where (("newer"."id" = "ontology"."ebook_version"."id") and ("newer"."version" > "ontology"."ebook_version"."version")))
view artefact_view
  select "ontology"."node_version"."id", "ontology"."node_version"."version", "ontology"."node_version"."action", "ontology"."node"."type", "ontology"."artefact_version"."label", "ontology"."node_version"."timestamp", "version_zero"."timestamp" as "created_at" from "ontology"."artefact_version" inner join "ontology"."node_version" on (("ontology"."artefact_version"."id" = "ontology"."node_version"."id") and ("ontology"."artefact_version"."version" = "ontology"."node_version"."version")) inner join "ontology"."node" on "ontology"."artefact_version"."id" = "ontology"."node"."id" inner join "ontology"."node_version" "version_zero" on (("ontology"."artefact_version"."id" = "version_zero"."id") and ("version_zero"."version" = 0)) where ((not exists (select 1 from "ontology"."node_version" "newer" where (("newer"."id" = "ontology"."artefact_version"."id") and ("newer"."version" > "ontology"."artefact_version"."version")))) and (case "ontology"."node"."type" when 'printed' then exists (select 1 from "ontology"."printed_version" where (("ontology"."artefact_version"."id" = "ontology"."printed_version"."id") and ("ontology"."artefact_version"."version" = "ontology"."printed_version"."version")) offset 0) when 'digital' then exists (select 1 from "ontology"."digital_version" where (("ontology"."artefact_version"."id" = "ontology"."digital_version"."id") and ("ontology"."artefact_version"."version" = "ontology"."digital_version"."version")) offset 0) when 'ebook' then ((exists (select 1 from "ontology"."printed_version" where (("ontology"."artefact_version"."id" = "ontology"."printed_version"."id") and ("ontology"."artefact_version"."version" = "ontology"."printed_version"."version")) offset 0)) and (exists (select 1 from "ontology"."digital_version" where (("ontology"."artefact_version"."id" = "ontology"."digital_version"."id") and ("ontology"."artefact_version"."version" = "ontology"."digital_version"."version")) offset 0)) and (exists (select 1 from "ontology"."ebook_version" where (("ontology"."artefact_version"."id" = "ontology"."ebook_version"."id") and ("ontology"."artefact_version"."version" = "ontology"."ebook_version"."version")) offset 0))) else false end))
