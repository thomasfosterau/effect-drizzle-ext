view book_view
  select "ontology"."node_version"."id", "ontology"."node_version"."version", "ontology"."node_version"."action", "ontology"."work_version"."title", "ontology"."work_version"."url", "ontology"."book_version"."pages", "ontology"."node_version"."timestamp", "version_zero"."timestamp" as "created_at" from "ontology"."book_version" inner join "ontology"."node_version" on (("ontology"."book_version"."id" = "ontology"."node_version"."id") and ("ontology"."book_version"."version" = "ontology"."node_version"."version")) inner join "ontology"."node" on "ontology"."book_version"."id" = "ontology"."node"."id" inner join "ontology"."work_version" on (("ontology"."book_version"."id" = "ontology"."work_version"."id") and ("ontology"."book_version"."version" = "ontology"."work_version"."version")) inner join "ontology"."node_version" "version_zero" on (("ontology"."book_version"."id" = "version_zero"."id") and ("version_zero"."version" = 0)) where not exists (select 1 from "ontology"."node_version" "newer" where (("newer"."id" = "ontology"."book_version"."id") and ("newer"."version" > "ontology"."book_version"."version")))
